<script>
    // Инициализация Telegram WebApp
    const tg = window.Telegram.WebApp;
    tg.ready();
    tg.expand();

    // Инициализация TON Connect UI
    // manifestUrl должен вести на реальный JSON-файл с метаданными вашего приложения!
    const tonConnectUI = new TON_CONNECT_UI.TonConnectUI({
        manifestUrl: 'https://githubusercontent.com',
        buttonRootId: 'ton-connect'
    });

    const balanceEl = document.getElementById('balance');
    const addressEl = document.getElementById('wallet-address');
    const sendSection = document.getElementById('send-section');

    // Отслеживание состояния подключения кошелька
    tonConnectUI.onStatusChange(async (wallet) => {
        if (wallet) {
            // Кошелек подключен. Получаем адрес (в формате Raw, переводим в Friendly-формат)
            const rawAddress = wallet.account.address;
            const friendlyAddress = convertToFriendly(rawAddress);
            
            addressEl.innerText = friendlyAddress;
            sendSection.style.display = 'flex';

            // Получаем баланс из блокчейна TON через публичное API
            fetchBalance(rawAddress);
        } else {
            // Кошелек отключен
            addressEl.innerText = "Кошелек не подключен";
            balanceEl.innerText = "0.00 TON";
            sendSection.style.display = 'none';
        }
    });

    // Функция запроса баланса (используем бесплатный API toncenter)
    async function fetchBalance(rawAddress) {
        try {
            const response = await fetch(`https://toncenter.com{rawAddress}`);
            const data = await response.json();
            if (data.ok) {
                const nanoGrams = data.result.balance;
                const tonBalance = (parseInt(nanoGrams) / 1000000000).toFixed(2);
                balanceEl.innerText = `${tonBalance} TON`;
            }
        } catch (e) {
            console.error("Ошибка при получении баланса", e);
        }
    }

    // Обработчик отправки транзакции
    document.getElementById('send-tx-btn').addEventListener('click', async () => {
        const recipient = document.getElementById('recipient').value;
        const amount = document.getElementById('amount').value;

        if (!recipient || !amount) {
            alert("Заполните адрес и сумму!");
            return;
        }

        // Переводим TON в Нанотоны (1 TON = 1,000,000,000 Нанотон)
        const nanoAmount = (parseFloat(amount) * 1000000000).toString();

        // Структура транзакции для TON Connect
        const transaction = {
            validUntil: Math.floor(Date.now() / 1000) + 60, // Срок жизни транзакции 60 сек
            messages: [
                {
                    address: recipient, 
                    amount: nanoAmount 
                }
            ]
        };

        try {
            tg.MainButton.setText('Ожидание подписи...').show();
            // Вызываем окно кошелька (например, Tonkeeper) внутри Telegram для подтверждения
            const result = await tonConnectUI.sendTransaction(transaction);
            
            tg.MainButton.setText('Транзакция отправлена!').setColor('#31b545');
            alert("Успешно! Хэш транзакции отправлен в сеть.");
            
            setTimeout(() => tg.MainButton.hide(), 3000);
        } catch (e) {
            console.error(e);
            tg.MainButton.hide();
            alert("Ошибка выполнения транзакции или пользователь отклонил её.");
        }
    });

    // Утилита для простого форматирования адреса для отображения (заглушка)
    function convertToFriendly(hexAddress) {
        return hexAddress.substring(0, 6) + '...' + hexAddress.substring(hexAddress.length - 6);
    }
